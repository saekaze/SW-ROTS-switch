# The SDK's two tools on Linux, with the vendored Zydis and toml++ (no network).
cmake_minimum_required(VERSION 3.20)
project(xbe_tools CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/../..)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/third_party/zydis zydis EXCLUDE_FROM_ALL)
file(GLOB RECOMP ${ROOT}/sdk/tools/recomp/*.cpp)
add_executable(xbe_recomp ${RECOMP})
target_include_directories(xbe_recomp PRIVATE ${ROOT}/sdk/runtime ${ROOT}/sdk/tools/recomp ${ROOT}/third_party/tomlplusplus/include)
target_link_libraries(xbe_recomp PRIVATE Zydis)
add_executable(xbe_symbols ${ROOT}/sdk/tools/symbols/main.cpp)
target_include_directories(xbe_symbols PRIVATE ${ROOT}/sdk/runtime ${ROOT}/third_party/tomlplusplus/include)
target_link_libraries(xbe_symbols PRIVATE Zydis)
